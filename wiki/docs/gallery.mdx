import useBaseUrl from '@docusaurus/useBaseUrl';

export const GalleryCard = ({ img, title, subtitle, tag, portrait }) => {
  const src = useBaseUrl(`/img/${img}`);
  return (
    <a
      className={`gallery-card${portrait ? ' gallery-card--mobile' : ''}`}
      href={src}
      target="_blank"
      rel="noopener"
    >
      <div className="gallery-card__media">
        <img src={src} alt={title} loading="lazy" />
      </div>
      <div className="gallery-card__body">
        <div className="gallery-card__title">
          {tag && <span className={`gallery-tag gallery-tag--${tag.toLowerCase()}`}>{tag}</span>}
          {title}
        </div>
        <div className="gallery-card__subtitle">{subtitle}</div>
      </div>
    </a>
  );
};
